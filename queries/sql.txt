-- eBay Philippines Database Setup
-- Run this script to create and seed the database

CREATE DATABASE IF NOT EXISTS ebayph;
USE ebayph;

-- Users Table
CREATE TABLE IF NOT EXISTS User (
    User_ID     CHAR(8) PRIMARY KEY,
    User_FName  VARCHAR(30) NOT NULL,
    User_LName  VARCHAR(30) NOT NULL,
    User_AccName VARCHAR(30) NOT NULL UNIQUE,
    User_Email  VARCHAR(50) NOT NULL UNIQUE,
    User_Password VARCHAR(255) NOT NULL,
    User_Contact CHAR(11) NOT NULL,
    User_Address VARCHAR(100) NOT NULL,
    User_DateReg DATE NOT NULL,
    User_AccType ENUM('buyer','seller','admin') NOT NULL DEFAULT 'buyer',
    User_Status  ENUM('active','suspended') NOT NULL DEFAULT 'active'
);

-- Seller Table
CREATE TABLE IF NOT EXISTS Seller (
    Sell_ID     CHAR(8) PRIMARY KEY,
    Sell_UserID CHAR(8) NOT NULL,
    Sell_Type   VARCHAR(20) NOT NULL,
    Sell_FeeRate DECIMAL(5,2) NOT NULL DEFAULT 0.00,
    Sell_Status  VARCHAR(10) NOT NULL DEFAULT 'Active',
    Sell_JoinDate DATE NOT NULL,
    FOREIGN KEY (Sell_UserID) REFERENCES User(User_ID) ON DELETE CASCADE
);

-- Business Table
CREATE TABLE IF NOT EXISTS Business (
    Bus_ID      CHAR(8) PRIMARY KEY,
    Bus_SellID  CHAR(8) NOT NULL,
    Bus_Name    VARCHAR(100) NOT NULL,
    Bus_TaxID   VARCHAR(30) NOT NULL,
    Bus_RegNum  VARCHAR(30) NOT NULL,
    Bus_Verified TINYINT(1) NOT NULL DEFAULT 0,
    Bus_VerDate  DATE,
    Bus_Type    VARCHAR(30) NOT NULL,
    Bus_Phone   CHAR(11) NOT NULL,
    Bus_Address VARCHAR(100) NOT NULL,
    FOREIGN KEY (Bus_SellID) REFERENCES Seller(Sell_ID) ON DELETE CASCADE
);

-- Category Table
CREATE TABLE IF NOT EXISTS Category (
    Cat_ID CHAR(8) PRIMARY KEY,
    Cat_Name VARCHAR(50) NOT NULL UNIQUE,
    Cat_Icon VARCHAR(50) DEFAULT 'bi-tag',
    Cat_Description VARCHAR(255),
    Cat_Status ENUM('active', 'inactive') DEFAULT 'active'
);


-- Product Table
CREATE TABLE IF NOT EXISTS Product (
    Prod_ID     CHAR(8) PRIMARY KEY,
    Prod_SellID CHAR(8) NOT NULL,
    Prod_CatID  CHAR(8),
    Prod_Title  VARCHAR(100) NOT NULL,
    Prod_Desc   VARCHAR(255) NOT NULL,
    Prod_Price  DECIMAL(10,2) NOT NULL,
    Prod_Image  VARCHAR(255) DEFAULT NULL,
    Prod_Stock  INT NOT NULL DEFAULT 0,
    Prod_Status VARCHAR(10) NOT NULL DEFAULT 'active',
    Prod_DateAdd DATE NOT NULL,
    Prod_DateUpd DATE,
    FOREIGN KEY (Prod_SellID) REFERENCES Seller(Sell_ID) ON DELETE CASCADE,
    FOREIGN KEY (Prod_CatID) REFERENCES Category(Cat_ID) ON DELETE CASCADE
);

-- Auction Table
CREATE TABLE IF NOT EXISTS Auction (
    Auc_ID      CHAR(8) PRIMARY KEY,
    Auc_ProdID  CHAR(8) NOT NULL,
    Auc_StartPrice DECIMAL(10,2) NOT NULL,
    Auc_HighBid  DECIMAL(10,2),
    Auc_StartDate DATETIME NOT NULL,
    Auc_EndDate  DATETIME NOT NULL,
    Auc_Status   VARCHAR(10) NOT NULL DEFAULT 'Active',
    FOREIGN KEY (Auc_ProdID) REFERENCES Product(Prod_ID) ON DELETE CASCADE
);

-- Bid Table
CREATE TABLE IF NOT EXISTS Bid (
    Bid_ID      CHAR(8) PRIMARY KEY,
    Bid_AucID   CHAR(8) NOT NULL,
    Bid_UserID  CHAR(8) NOT NULL,
    Bid_Amount  DECIMAL(10,2) NOT NULL,
    Bid_Date    DATETIME NOT NULL,
    FOREIGN KEY (Bid_AucID) REFERENCES Auction(Auc_ID) ON DELETE CASCADE,
    FOREIGN KEY (Bid_UserID) REFERENCES User(User_ID) ON DELETE CASCADE
);


-- Cart Table
CREATE TABLE IF NOT EXISTS Cart (
    Cart_UserID CHAR(8) NOT NULL,
    Cart_ProdID CHAR(8) NOT NULL,
    Cart_Qty    INT NOT NULL DEFAULT 1,
    Cart_DateAdd DATE NOT NULL,
    Cart_DateUpd DATE,
    PRIMARY KEY (Cart_UserID, Cart_ProdID),
    FOREIGN KEY (Cart_UserID) REFERENCES User(User_ID) ON DELETE CASCADE,
    FOREIGN KEY (Cart_ProdID) REFERENCES Product(Prod_ID) ON DELETE CASCADE
);

-- Wishlist Table
CREATE TABLE IF NOT EXISTS Wishlist (
    Wish_UserID CHAR(8) NOT NULL,
    Wish_ProdID CHAR(8) NOT NULL,
    Wish_DateAdd DATE NOT NULL,
    Wish_PriceAdd DECIMAL(10,2) NOT NULL,
    Wish_Notified TINYINT(1) NOT NULL DEFAULT 0,
    PRIMARY KEY (Wish_UserID, Wish_ProdID),
    FOREIGN KEY (Wish_UserID) REFERENCES User(User_ID) ON DELETE CASCADE,
    FOREIGN KEY (Wish_ProdID) REFERENCES Product(Prod_ID) ON DELETE CASCADE
);

-- Order Table
CREATE TABLE IF NOT EXISTS `Order` (
    Order_ID    CHAR(8) PRIMARY KEY,
    Order_UserID CHAR(8) NOT NULL,
    Order_Date  DATE NOT NULL,
    Order_Total  DECIMAL(10,2) NOT NULL,
    Order_PayStat VARCHAR(20) NOT NULL DEFAULT 'Pending',
    Order_ShipAdd VARCHAR(100) NOT NULL,
    Order_TrackNum VARCHAR(30),
    FOREIGN KEY (Order_UserID) REFERENCES User(User_ID) ON DELETE CASCADE
);

-- Order Item Table
CREATE TABLE IF NOT EXISTS OrderItem (
    Item_ID     CHAR(8) PRIMARY KEY,
    Item_OrderID CHAR(8) NOT NULL,
    Item_ProdID  CHAR(8) NOT NULL,
    Item_Qty    INT NOT NULL,
    Item_Price  DECIMAL(10,2) NOT NULL,
    Item_Sub    DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (Item_OrderID) REFERENCES `Order`(Order_ID) ON DELETE CASCADE,
    FOREIGN KEY (Item_ProdID) REFERENCES Product(Prod_ID) ON DELETE CASCADE
);

-- Shipment Table
CREATE TABLE IF NOT EXISTS Shipment (
    Ship_ID     CHAR(8) PRIMARY KEY,
    Ship_OrderID CHAR(8) NOT NULL,
    Ship_Status  VARCHAR(20) NOT NULL DEFAULT 'Shipped',
    Ship_PickupDate DATE,
    Ship_TransitDate DATE,
    Ship_DelivDate DATE,
    Ship_Location VARCHAR(100),
    FOREIGN KEY (Ship_OrderID) REFERENCES `Order`(Order_ID) ON DELETE CASCADE
);

-- Feedback Table
CREATE TABLE IF NOT EXISTS Feedback (
    Feed_ID     CHAR(8) PRIMARY KEY,
    Feed_OrderID CHAR(8) NOT NULL,
    Feed_SellID  CHAR(8) NOT NULL,
    Feed_UserID  CHAR(8) NOT NULL,
    Feed_Rating  INT NOT NULL,
    Feed_Comment VARCHAR(200),
    Feed_Date   DATE NOT NULL,
    FOREIGN KEY (Feed_OrderID) REFERENCES `Order`(Order_ID) ON DELETE CASCADE,
    FOREIGN KEY (Feed_SellID) REFERENCES Seller(Sell_ID) ON DELETE CASCADE,
    FOREIGN KEY (Feed_UserID) REFERENCES User(User_ID) ON DELETE CASCADE
);

-- Admin Table
CREATE TABLE IF NOT EXISTS Admin (
    Admin_ID    CHAR(8) PRIMARY KEY,
    Admin_UserID CHAR(8) NOT NULL,
    Admin_Level  INT NOT NULL DEFAULT 1,
    Admin_Perm  VARCHAR(100) NOT NULL,
    Admin_LastLog DATETIME,
    FOREIGN KEY (Admin_UserID) REFERENCES User(User_ID) ON DELETE CASCADE
);
